{{- $link := .Destination -}} {{- $isRemote := strings.HasPrefix $link "http" -}} {{- $isInternalLink := false -}} {{- if $isRemote -}} {{- /* Handle absolute links to this site or the public docs host (resolve at build time). The public host is hardcoded because every site imports the public docs module, so those pages always exist locally. The dynamic baseURL check handles the internal site resolving its own internal-documentation.… links. */ -}} {{- $baseHost := printf "https://%s" (urls.Parse site.BaseURL).Host -}} {{- $isHummingbirdLink := or (strings.HasPrefix $link $baseHost) (strings.HasPrefix $link "https://hummingbird-project.io") -}} {{- if $isHummingbirdLink -}} {{- $isInternalLink = true -}} {{- $url := urls.Parse .Destination -}} {{- $fragment := "" -}} {{- with $url.Fragment }}{{ $fragment = printf "#%s" . }}{{ end -}} {{- /* Try /l/ alias lookup if path starts with /l/ */ -}} {{- if strings.HasPrefix $url.Path "/l/" -}} {{- $urlPath := strings.TrimSuffix "/" $url.Path -}} {{- $foundPage := false -}} {{- range $page := site.Pages -}} {{- if not $foundPage -}} {{- range $page.Params.aliases -}} {{- if or (eq . $urlPath) (eq . $url.Path) -}} {{- $link = printf "%s%s" $page.RelPermalink $fragment -}} {{- $foundPage = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- else if in (slice "" ".md") (path.Ext $url.Path) -}} {{- /* Try direct path lookup; skip static files such as .html and .pdf */ -}} {{- with .Page.GetPage $url.Path -}} {{- $link = printf "%s%s" .RelPermalink $fragment -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $url := urls.Parse .Destination -}} {{- if and $url.Path (in (slice "" ".md") (path.Ext $url.Path)) -}} {{- $fragment := "" -}} {{- with $url.Fragment }}{{ $fragment = printf "#%s" . }}{{ end -}} {{- with .Page.GetPage $url.Path }}{{ $link = printf "%s%s" .RelPermalink $fragment }}{{ end -}} {{- end -}} {{- end -}} {{- print "" -}} {{- .Text | safeHTML -}} {{- print "" -}}
